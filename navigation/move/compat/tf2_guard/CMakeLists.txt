cmake_minimum_required(VERSION 3.16)
project(nav_tf2_guard)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(tf2_ros REQUIRED)
if(NOT tf2_ros_VERSION VERSION_EQUAL "0.25.23")
  message(FATAL_ERROR "This compatibility guard targets tf2_ros 0.25.23; revalidate before using another version.")
endif()
add_library(nav_tf2_guard SHARED tf2_guard.cpp)
ament_target_dependencies(nav_tf2_guard tf2_ros)
target_link_libraries(nav_tf2_guard dl)
add_executable(test_tf2_guard test_tf2_guard.cpp)
ament_target_dependencies(test_tf2_guard tf2_ros)
